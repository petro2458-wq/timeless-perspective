/** * generate-daily-news.js * * Called by GitHub Actions at 4 AM and 10 AM ET (Mon-Fri). * Fetches the FREE ForexFactory JSON calendar feed and generates * a styled daily-news.html — NO Claude API needed. $0/month. * * Data source: https://nfs.faireconomy.media/ff_calendar_thisweek.json * Rate limit: 2 requests per 5 minutes (we only call twice/day) */ const fs = require('fs'); const https = require('https'); // ── DATE HELPERS ────────────────────────────────────────────────────────── const now = new Date(); const etOptions = { timeZone: 'America/New_York' }; const todayFormatted = now.toLocaleDateString('en-US', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric', ...etOptions }); const hour = now.toLocaleString('en-US', { hour: 'numeric', hour12: true, ...etOptions }); const etNow = new Date(now.toLocaleString('en-US', etOptions)); const todayDateStr = etNow.toISOString().split('T')[0]; // YYYY-MM-DD in ET function getWeekDates(offsetWeeks = 0) { const dow = etNow.getDay(); const daysToMonday = (dow === 0 ? 6 : dow - 1); const mondayMs = etNow.getTime() - (daysToMonday * 86400000) + (offsetWeeks * 7 * 86400000); const labels = ['MON','TUE','WED','THU','FRI']; return labels.map((lbl, i) => { const d = new Date(mondayMs + i * 86400000); const etStr = d.toLocaleString('en-US', { ...etOptions, month: 'numeric', day: 'numeric' }); const [m, day] = etStr.split(',')[0].split('/'); return { label: lbl, date: `${m}/${day}`, fullDate: d.toISOString().split('T')[0], ms: mondayMs + i * 86400000 }; }); } const thisWeek = getWeekDates(0); const nextWeek = getWeekDates(1); const todayDow = etNow.getDay(); // 0=Sun ... 6=Sat // ── FETCH HELPER ────────────────────────────────────────────────────────── function fetchJSON(url) { return new Promise((resolve, reject) => { https.get(url, { headers: { 'User-Agent': 'TimelessPerspective/1.0' } }, (res) => { let data = ''; res.on('data', chunk => data += chunk); res.on('end', () => { try { resolve(JSON.parse(data)); } catch (e) { reject(new Error(`JSON parse failed: ${e.message} — raw: ${data.substring(0, 200)}`)); } }); }).on('error', reject); }); } // ── FORMAT HELPERS ──────────────────────────────────────────────────────── function formatTime(dateStr) { const d = new Date(dateStr); let h = d.getHours(); const m = d.getMinutes(); const ampm = h >= 12 ? 'PM' : 'AM'; h = h % 12 || 12; return m === 0 ? `${h}:00 ${ampm} ET` : `${h}:${String(m).padStart(2,'0')} ${ampm} ET`; } function formatTimeShort(dateStr) { const d = new Date(dateStr); let h = d.getHours(); const m = d.getMinutes(); const ampm = h >= 12 ? 'p' : 'a'; h = h % 12 || 12; return m === 0 ? `${h}${ampm}` : `${h}:${String(m).padStart(2,'0')}${ampm}`; } function impactClass(impact) { if (impact === 'High') return 'badge-critical'; if (impact === 'Medium') return 'badge-high'; return 'badge-low'; } function impactLabel(impact) { if (impact === 'High') return 'HIGH IMPACT'; if (impact === 'Medium') return 'MEDIUM'; return 'LOW'; } // ── MAIN ────────────────────────────────────────────────────────────────── async function generateCalendar() { console.log(`🗞️ Generating Daily News for ${todayFormatted} (${hour} ET run)...`); // Fetch ForexFactory calendar let allEvents; try { allEvents = await fetchJSON('https://nfs.faireconomy.media/ff_calendar_thisweek.json'); console.log(`✅ Fetched ${allEvents.length} events from ForexFactory feed`); } catch (err) { // Fallback to CDN version try { allEvents = await fetchJSON('https://cdn-nfs.faireconomy.media/ff_calendar_thisweek.json'); console.log(`✅ Fetched ${allEvents.length} events from CDN fallback`); } catch (err2) { throw new Error(`Both FF feeds failed: ${err.message} / ${err2.message}`); } } // Filter USD events with High or Medium impact const usdEvents = allEvents.filter(e => e.country === 'USD' && (e.impact === 'High' || e.impact === 'Medium')); console.log(`📊 ${usdEvents.length} USD high/medium impact events this week`); // Parse dates and group by day const eventsByDate = {}; for (const e of usdEvents) { const d = new Date(e.date); const dateKey = d.toLocaleDateString('en-US', { ...etOptions, year: 'numeric', month: '2-digit', day: '2-digit' }); if (!eventsByDate[dateKey]) eventsByDate[dateKey] = []; eventsByDate[dateKey].push({ ...e, parsedDate: d }); } // Today's events const todayKey = etNow.toLocaleDateString('en-US', { ...etOptions, year: 'numeric', month: '2-digit', day: '2-digit' }); const todayEvents = (eventsByDate[todayKey] || []).sort((a, b) => a.parsedDate - b.parsedDate); console.log(`📅 ${todayEvents.length} USD events today`); // Group today's events by time const timeGroups = {}; for (const e of todayEvents) { const timeKey = formatTime(e.date); if (!timeGroups[timeKey]) timeGroups[timeKey] = []; timeGroups[timeKey].push(e); } // Build week strip notes (red folder events with times for each day) function getWeekDayNote(weekDay) { const dayDate = new Date(weekDay.ms); const dayKey = dayDate.toLocaleDateString('en-US', { ...etOptions, year: 'numeric', month: '2-digit', day: '2-digit' }); const dayEvents = (eventsByDate[dayKey] || []).sort((a, b) => new Date(a.date) - new Date(b.date)); const highEvents = dayEvents.filter(e => e.impact === 'High'); const medEvents = dayEvents.filter(e => e.impact === 'Medium'); if (highEvents.length > 0) { return highEvents.map(e => { const shortName = e.title.replace(/ m\/m| y\/y| q\/q| ytd\/y/gi, '').replace(/Preliminary |Final |Flash |Revised /gi, '').trim(); return `${shortName} ${formatTimeShort(e.date)}`; }).join(' · '); } else if (medEvents.length > 0) { return medEvents.slice(0, 2).map(e => { const shortName = e.title.replace(/ m\/m| y\/y| q\/q| ytd\/y/gi, '').replace(/Preliminary |Final |Flash |Revised /gi, '').trim(); return `${shortName} ${formatTimeShort(e.date)}`; }).join(' · '); } return 'No high impact'; } // ── BUILD HTML ──────────────────────────────────────────────────────── const isWeekend = todayDow === 0 || todayDow === 6; // Today's time cards let timeCardsHtml = ''; if (todayEvents.length === 0 && !isWeekend) { timeCardsHtml = `
No High-Impact USD Events Today
Lower-volatility session expected. Focus on technicals and price action.
`; } else if (isWeekend) { timeCardsHtml = `
Weekend — Markets Closed
Use this time to review your week, journal your trades, and prepare for next week's events below.
`; } else { for (const [time, events] of Object.entries(timeGroups)) { const maxImpact = events.some(e => e.impact === 'High') ? 'High' : 'Medium'; const dataRows = events.map(e => { const hasActual = e.actual && e.actual.trim() !== ''; const actualClass = hasActual ? 'val-actual' : ''; return `
${e.title} A:${hasActual ? e.actual : 'Pending'} F:${e.forecast || '—'} P:${e.previous || '—'}
`; }).join(''); timeCardsHtml += `
${time} ${impactLabel(maxImpact)}
${dataRows}
`; } } // Week strip const weekStripHtml = thisWeek.map(day => { const isToday = day.date === `${etNow.getMonth()+1}/${etNow.getDate()}`; const isPast = day.ms < (etNow.getTime() - 86400000); return `
${day.label} ${day.date} ${isPast ? '✓ Done' : getWeekDayNote(day)}
`; }).join(''); // Next week strip const nextWeekHtml = nextWeek.map(day => `
${day.label} ${day.date} ${getWeekDayNote(day)}
`).join(''); // Alert box const highToday = todayEvents.filter(e => e.impact === 'High'); let alertHtml = ''; if (isWeekend) { alertHtml = `
📅 Weekend Preview — Markets reopen Sunday 6 PM ET. Review next week's calendar below.
`; } else if (highToday.length > 0) { const alertItems = highToday.map(e => `${e.title} at ${formatTimeShort(e.date)}`).join(', '); alertHtml = `
🔴 RED FOLDER TODAY: ${alertItems}. Expect elevated volatility around these releases.
`; } else if (todayEvents.length > 0) { alertHtml = `
🟠 MEDIUM IMPACT: ${todayEvents.length} scheduled USD release${todayEvents.length > 1 ? 's' : ''} today. Moderate volatility expected.
`; } else { alertHtml = `
✅ CLEAN SESSION: No high-impact USD events scheduled. Favorable for technical setups.
`; } // Count events const totalHighThisWeek = usdEvents.filter(e => e.impact === 'High').length; const totalMedThisWeek = usdEvents.filter(e => e.impact === 'Medium').length; // Full HTML output const html = `

📅 USD Economic Calendar

${todayFormatted} — ${hour} ET

${alertHtml}
📊 This Week: ${totalHighThisWeek} red folder + ${totalMedThisWeek} orange folder USD events. Data sourced from ForexFactory.
${weekStripHtml}
${timeCardsHtml}

📅 Next Week — ${nextWeek[0].label} ${nextWeek[0].date} – ${nextWeek[4].label} ${nextWeek[4].date}

${nextWeekHtml}
USD Economic Calendar • ${todayFormatted} • Source: ForexFactory • For informational purposes only — not investment advice
`; // Write output const output = `\n\n${html}`; fs.writeFileSync('daily-news.html', output, 'utf-8'); console.log(`✅ Written daily-news.html (${(output.length / 1024).toFixed(1)} KB) — ${todayEvents.length} events today, ${usdEvents.length} this week`); } generateCalendar().catch(err => { console.error('❌ Generation failed:', err.message); const fallback = `
📡

Daily News Updating...

Calendar is being refreshed. Check back shortly.

Last attempt: ${todayFormatted} ${hour} ET

Error: ${err.message}

`; fs.writeFileSync('daily-news.html', fallback, 'utf-8'); process.exit(1); });
